# ============================================================
# raw_log.py · M2-C · 原始日志
# 按 round_id 存储每一轮的完整原文
# ============================================================

from typing import Dict, List, Any, Optional


class RawLog:
    """原始日志：按 round_id 存每一轮的完整原文。

    key = round_id (int)
    value = 那一轮的完整记录 dict（由调用方保证结构）
    """

    def __init__(self):
        self._records: Dict[int, Dict[str, Any]] = {}

    def append(self, round_data: Dict[str, Any]) -> Optional[Dict[str, Any]]:
        """追加一轮。若 round_id 已存在，覆盖并返回旧值；否则返回 None。"""
        rid = round_data.get("round_id")
        if rid is None:
            return None
        old = self._records.get(rid)
        self._records[rid] = round_data
        return old

    def get(self, round_id: int) -> Optional[Dict[str, Any]]:
        """按 round_id 取原文，无则返回 None。"""
        return self._records.get(round_id)

    def get_many(self, round_ids: List[int]) -> List[Optional[Dict[str, Any]]]:
        """按 round_id 列表批量取，返回列表（顺序与入参一致，缺的返回 None）。"""
        return [self._records.get(rid) for rid in round_ids]

    def __len__(self) -> int:
        return len(self._records)

    def all_ids(self) -> List[int]:
        """返回所有 round_id 的列表（升序）。"""
        return sorted(self._records.keys())

# ============================================================
# archiver.py · M2-C · 归档器（带桩 + RawLog 写入）
# 把短期缓冲区吐出的记录整理成 MemoryPackage，并写入原始日志
# ============================================================

from typing import List, Dict, Any, Optional

from schemas import MemoryPackage
from short_term_buffer import ShortTermBuffer
from raw_log import RawLog


class Archiver:
    """归档器：将短期缓冲区的记录整理成 MemoryPackage。

    M2-C：接入 RawLog，归档时把 flushed 记录同步写入 raw_log。
    organize_narrative 仍为桩，不接 memory_graph，不接 LLM。
    """

    def __init__(self, buffer: ShortTermBuffer, raw_log: RawLog, threshold: int = 6):
        self._buffer = buffer
        self._raw_log = raw_log
        self._threshold = threshold
        self._packages: List[MemoryPackage] = []

    def should_archive(self) -> bool:
        """判断是否该归档：缓冲区长度 >= 阈值时返回 True。"""
        return len(self._buffer) >= self._threshold

    def archive(self) -> Optional[MemoryPackage]:
        """执行归档流程。返回生成的 MemoryPackage，若无可归档内容返回 None。"""
        flushed, _ = self._buffer.flush_keep_recent(2)
        if not flushed:
            return None

        # 把 flushed 里每一轮写进 raw_log
        for record in flushed:
            self._raw_log.append(record)

        pkg = self.organize_narrative(flushed)
        self._packages.append(pkg)
        return pkg

    def organize_narrative(self, records: List[Dict[str, Any]]) -> MemoryPackage:
        # 【桩】此处未来替换为 LLM 整理，当前为占位实现
        seq = len(self._packages) + 1
        round_ids = [str(r.get("round_id", "")) for r in records]
        raw_pointer = "-".join(round_ids)

        # 关键词：取前 2 条记录 user_input 的前 4 个字
        keywords = []
        for r in records[:2]:
            text = r.get("user_input", "")
            if text:
                keywords.append(text[:4])
        keywords = keywords[:3]

        # 叙事字段：桩占位
        first_user = records[0].get("user_input", "") if records else ""
        last_user = records[-1].get("user_input", "") if records else ""
        count = len(records)

        narrative_cause = f"桩：起因是用户在第{round_ids[0]}轮说'{first_user}'"
        narrative_trigger = f"桩：触发点是{count}轮对话积累"
        narrative_consequence = f"桩：后果是到第{round_ids[-1]}轮用户说'{last_user}'"
        narrative_evaluation = f"桩：我的评价是中性，共{count}轮"
        narrative_emotion = "桩：我的情绪是 neutral（占位）"
        narrative_thought = f"桩：我的想法是这{count}轮可以归纳为一段记忆"

        return MemoryPackage(
            seq=seq,
            emotion_value=0.5,
            emotion_polarity="neutral",
            keywords=keywords,
            raw_pointer=raw_pointer,
            narrative_cause=narrative_cause,
            narrative_trigger=narrative_trigger,
            narrative_consequence=narrative_consequence,
            narrative_evaluation=narrative_evaluation,
            narrative_emotion=narrative_emotion,
            narrative_thought=narrative_thought,
        )

    def get_packages(self) -> List[MemoryPackage]:
        """返回全部已归档的 MemoryPackage。"""
        return list(self._packages)

# ============================================================
# decompressor.py · M2-C · 解压器
# 从 MemoryPackage 按不同器官的需求取出对应信息
# ============================================================

from typing import List, Dict, Any

from schemas import MemoryPackage
from raw_log import RawLog


class Decompressor:
    """解压器：从 MemoryPackage 按器官需求取出信息。

    字段来源：
      - MemoryPackage 直接字段（seq, emotion_value, narrative_* 等）
      - raw_pointer 解析出的轮次 → 去 RawLog 取原文
    """

    def __init__(self, raw_log: RawLog):
        self._raw_log = raw_log

    # ---- 解析 raw_pointer ----

    def _parse_raw_pointer(self, raw_pointer: str) -> List[int]:
        """解析 raw_pointer（"1-2-3" 格式），返回 round_id 列表。
        解析失败时返回空列表。"""
        if not raw_pointer:
            return []
        parts = raw_pointer.split("-")
        ids = []
        for p in parts:
            p = p.strip()
            if not p:
                return []
            try:
                ids.append(int(p))
            except ValueError:
                return []
        return ids

    def _get_raw(self, pkg: MemoryPackage) -> List[Dict[str, Any]]:
        ids = self._parse_raw_pointer(pkg.raw_pointer)
        if not ids:
            return []
        results = self._raw_log.get_many(ids)
        # 过滤掉 None（缺失的轮次），但保持顺序
        return [r for r in results if r is not None]

    # ---- 器官专用方法 ----

    def for_emotion_evaluator(self, pkg: MemoryPackage) -> Dict[str, Any]:
        return {
            "emotion_value": pkg.emotion_value,
            "emotion_polarity": pkg.emotion_polarity,
        }

    def for_danger_evaluator(self, pkg: MemoryPackage) -> Dict[str, Any]:
        return {
            "consequence": pkg.narrative_consequence,
            "evaluation": pkg.narrative_evaluation,
            "raw": self._get_raw(pkg),
        }

    def for_policy(self, pkg: MemoryPackage) -> Dict[str, Any]:
        return {
            "consequence": pkg.narrative_consequence,
            "thought": pkg.narrative_thought,
            "raw": self._get_raw(pkg),
        }

    def for_xinguang(self, pkg: MemoryPackage) -> Dict[str, Any]:
        return {
            "keywords": pkg.keywords,
            "emotion_value": pkg.emotion_value,
        }

    # ---- 通用方法 ----

    def get_fields(self, pkg: MemoryPackage, fields: List[str]) -> Dict[str, Any]:
        """按字段名列表取，返回 dict。

        特殊字段 "raw"：返回该压缩包对应的全量原文列表。
        其他字段：直接从 MemoryPackage 上取属性，不存在则返回 None。
        """
        result = {}
        for f in fields:
            if f == "raw":
                result[f] = self._get_raw(pkg)
            else:
                result[f] = getattr(pkg, f, None)
        return result

import sys
import os
sys.path.insert(0, os.path.join(os.path.dirname(__file__), ".."))

from short_term_buffer import ShortTermBuffer
from raw_log import RawLog
from archiver import Archiver
from decompressor import Decompressor


def make_round(rid, user, ai, ts):
    return {
        "round_id": rid,
        "user_input": user,
        "ai_output": ai,
        "timestamp": ts,
    }


def check(label, condition, detail=""):
    if condition:
        print(f"  {label} ✓")
        return True
    print(f"  {label} ✗  {detail}")
    return False


def main():
    buf = ShortTermBuffer(default_keep=2)
    raw_log = RawLog()
    archiver = Archiver(buf, raw_log, threshold=6)
    decomp = Decompressor(raw_log)

    # 1. 追加 7 轮
    dialogs = [
        (1, "你好啊", "你好，有什么可以帮你的？"),
        (2, "今天心情怎么样", "还不错，就正常吧"),
        (3, "你这个做的不行", "……抱歉，我再看看"),
        (4, "算了你也做不好", "我会努力改进的"),
        (5, "上次那个问题", "哪个问题呀？"),
        (6, "就是代码那个", "哦，我记着呢"),
        (7, "你记得挺清楚嘛", "哈哈谢谢"),
    ]
    for rid, user, ai in dialogs:
        buf.append(make_round(rid, user, ai, 1000.0 + rid))

    print(f"=== 追加 7 轮，buffer 长度 = {len(buf)} ===")

    # 2. 归档
    print()
    print("=== 执行归档 ===")
    pkg = archiver.archive()
    if pkg is None:
        print("TODO_REVIEW: archive 返回 None")
        return 1
    print(f"  seq = {pkg.seq}")
    print(f"  raw_pointer = {pkg.raw_pointer}")
    print(f"  buffer 剩余 = {len(buf)} 轮")

    # 3. 检查 raw_log 里有 5 轮
    print()
    print("=== raw_log 检查 ===")
    print(f"  raw_log 长度 = {len(raw_log)}")
    print(f"  all_ids = {raw_log.all_ids()}")

    all_ok = True
    all_ok &= check("raw_log 有 5 轮", len(raw_log) == 5, f"实际 = {len(raw_log)}")

    # 4. 四个器官专用方法

    r = decomp.for_emotion_evaluator(pkg)
    print(f"  emotion_value    = {r['emotion_value']}")
    print(f"  emotion_polarity = {r['emotion_polarity']}")
    all_ok &= check("有 emotion_value", "emotion_value" in r)
    all_ok &= check("有 emotion_polarity", "emotion_polarity" in r)

    print()
    print("=== for_danger_evaluator ===")
    r = decomp.for_danger_evaluator(pkg)
    print(f"  consequence      = {r['consequence']}")
    print(f"  evaluation       = {r['evaluation']}")
    print(f"  raw 条数         = {len(r['raw'])}")
    all_ok &= check("有 consequence", "consequence" in r)
    all_ok &= check("有 evaluation", "evaluation" in r)
    all_ok &= check("raw 有 5 条", len(r["raw"]) == 5, f"实际 = {len(r['raw'])}")

    print()
    print("=== for_policy ===")
    r = decomp.for_policy(pkg)
    print(f"  consequence      = {r['consequence']}")
    print(f"  thought          = {r['thought']}")
    print(f"  raw 条数         = {len(r['raw'])}")
    all_ok &= check("有 consequence", "consequence" in r)
    all_ok &= check("有 thought", "thought" in r)
    all_ok &= check("raw 有 5 条", len(r["raw"]) == 5, f"实际 = {len(r['raw'])}")

    print()
    print("=== for_xinguang ===")
    r = decomp.for_xinguang(pkg)
    print(f"  keywords         = {r['keywords']}")
    print(f"  emotion_value    = {r['emotion_value']}")
    all_ok &= check("有 keywords", "keywords" in r)
    all_ok &= check("有 emotion_value", "emotion_value" in r)

    # 5. get_fields 通用方法
    print()
    print("=== get_fields(pkg, ['emotion_value', 'raw']) ===")
    r = decomp.get_fields(pkg, ["emotion_value", "raw"])
    print(f"  emotion_value    = {r['emotion_value']}")
    print(f"  raw 条数         = {len(r['raw'])}")
    all_ok &= check("emotion_value 正确", r["emotion_value"] == 0.5, f"实际 = {r['emotion_value']}")
    all_ok &= check("raw 返回 5 条原文", len(r["raw"]) == 5, f"实际 = {len(r['raw'])}")

    # 6. 解压器返回的 raw 内容与 raw_log 里存的完全一致
    print()
    print("=== raw 内容一致性检查 ===")
    danger_raw = decomp.for_danger_evaluator(pkg)["raw"]
    log_records = raw_log.get_many([1, 2, 3, 4, 5])
    match = True
    for i in range(5):
        if danger_raw[i] is not log_records[i]:
            # 不是同一个对象没关系，比较内容
            if danger_raw[i]["round_id"] != log_records[i]["round_id"]:
                match = False
                break
            if danger_raw[i]["user_input"] != log_records[i]["user_input"]:
                match = False
                break
    all_ok &= check("解压器 raw 与 raw_log 内容一致", match)

    # 7. 解析失败场景：空 raw_pointer 返回空列表
    print()
    print("=== 解析失败场景 ===")
    from schemas import MemoryPackage
    bad_pkg = MemoryPackage(raw_pointer="")
    bad_raw = decomp.for_danger_evaluator(bad_pkg)["raw"]
    all_ok &= check("空 raw_pointer 返回空列表", bad_raw == [], f"实际 = {bad_raw}")

    bad_pkg2 = MemoryPackage(raw_pointer="abc-xyz")
    bad_raw2 = decomp.for_danger_evaluator(bad_pkg2)["raw"]
    all_ok &= check("非法格式返回空列表", bad_raw2 == [], f"实际 = {bad_raw2}")

    print()
    if all_ok:
        print("探针完成，全部断言通过")
        return 0
    else:
        print("TODO_REVIEW: 断言失败")
        return 1


if __name__ == "__main__":
    sys.exit(main())
