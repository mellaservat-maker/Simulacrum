6. embedder.py 完整原文

# ============================================================
# embedder.py · V4.5 · 可插拔嵌入器（M3-T3 第一阶段）
# 只新建，不改动任何现有 .py 文件
# 接口：
#   get_embedder(name="bge") 工厂函数
#   每个实例有 dim 属性与 encode(texts) 方法
#   HashEmbedder：复刻 event_parser.hash_to_8d 的 8 维 hash
#   BGEEmbedder ：BGE-Large-zh-v1.5，1024 维，懒加载，GPU 优先
# 默认后端：BGE
# ============================================================

import hashlib

import numpy as np


class HashEmbedder:
    """复刻 event_parser.hash_to_8d 的 8 维 hash 嵌入，输出必须完全一致。"""

    dim = 8

    def encode(self, texts):
        single = isinstance(texts, str)
        items = [texts] if single else list(texts)
        vecs = np.stack([self._hash_to_8d(t) for t in items])
        return vecs[0] if single else vecs

    @staticmethod
    def _hash_to_8d(raw_text):
        h = hashlib.md5(raw_text.encode()).digest()
        vec = np.array([h[i] / 255.0 for i in range(8)])
        return vec / (np.linalg.norm(vec) + 1e-8)


class BGEEmbedder:
    """BGE-Large-zh-v1.5 语义嵌入，1024 维，懒加载，GPU 优先。"""

    dim = 1024
    MODEL_NAME = "BAAI/bge-large-zh-v1.5"

    def __init__(self, model_name=None, device=None):
        self.model_name = model_name or self.MODEL_NAME
        self.device = device
        self._model = None

    def _load(self):
        if self._model is not None:
            return
        import torch
        from sentence_transformers import SentenceTransformer

        if self.device is None:
            self.device = "cuda" if torch.cuda.is_available() else "cpu"
        self._model = SentenceTransformer(self.model_name, device=self.device)

    def encode(self, texts):
        self._load()
        single = isinstance(texts, str)
        items = [texts] if single else list(texts)
        vecs = self._model.encode(
            items, normalize_embeddings=True, convert_to_numpy=True
        )
        return vecs[0] if single else vecs


def get_embedder(name="bge"):
    """工厂函数，默认后端为 BGE。"""
    key = (name or "bge").lower()
    if key in ("bge", "bge-large", "bge-large-zh-v1.5"):
        return BGEEmbedder()
    if key in ("hash", "hash8", "hash_to_8d"):
        return HashEmbedder()
    raise ValueError(f"Unknown embedder name: {name}")

# ============================================================
# scripts/probe_embedder.py · M3-T3 第一阶段探针
# 验证可插拔 embedder：维度、语义区分度、hash 兼容性
# ============================================================

import os
import sys

import numpy as np

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
if ROOT not in sys.path:
    sys.path.insert(0, ROOT)

from embedder import get_embedder
from event_parser import hash_to_8d


def cosine(a, b):
    a = np.asarray(a, dtype=float)
    b = np.asarray(b, dtype=float)
    return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b) + 1e-12))


def main():
    bge = get_embedder("bge")
    hash_emb = get_embedder("hash")

    print("=== 1. 维度 ===")
    print("get_embedder('bge').dim  =", bge.dim)
    print("get_embedder('hash').dim =", hash_emb.dim)

    t1 = "你真蠢"
    t2 = "你太笨了"
    t3 = "今天天气真好"

    print()
    print("=== 2. 编码三组文字 ===")
    v1 = bge.encode(t1)
    v2 = bge.encode(t2)
    v3 = bge.encode(t3)
    print("encode('%s') shape = %s" % (t1, np.asarray(v1).shape))
    print("encode('%s') shape = %s" % (t2, np.asarray(v2).shape))
    print("encode('%s') shape = %s" % (t3, np.asarray(v3).shape))

    print()
    print("=== 3. 两两 cosine ===")
    c_near = cosine(v1, v2)
    c_far = cosine(v1, v3)
    print("cosine(%s, %s) = %.6f   (预期 > 0.7)" % (t1, t2, c_near))
    print("cosine(%s, %s) = %.6f   (预期 < 0.5)" % (t1, t3, c_far))

    print()
    print("=== 4. 断言 ===")
    assert c_near > 0.7, "近义对相似度应 > 0.7，实际 %.6f" % c_near
    assert c_far < 0.5, "远义对相似度应 < 0.5，实际 %.6f" % c_far
    assert c_near > c_far, "近义对相似度应大于远义对"
    print("近义 > 0.7 : PASS")
    print("远义 < 0.5 : PASS")
    print("前者 > 后者 : PASS")

    print()
    print("=== 5. hash 兼容性（逐元素一致）===")
    h_now = np.asarray(hash_emb.encode(t1))
    h_ref = np.asarray(hash_to_8d(t1))
    print("embedder.encode('%s') = %s" % (t1, h_now.tolist()))
    print("event_parser.hash_to_8d('%s') = %s" % (t1, h_ref.tolist()))
    assert np.array_equal(h_now, h_ref), "hash embedder 与 event_parser.hash_to_8d 不一致"
    print("逐元素完全一致 : PASS")

    print()
    print("全部断言通过")


if __name__ == "__main__":
    main()

12. 过程中遇到的问题与解决

问题 1：默认 huggingface.co 被拦截
现象：探针首次运行报 WinError 10061 目标计算机积极拒绝，反复重试 5 次仍失败。
原因：huggingface.co 直连不可达；系统代理 127.0.0.1:7890 未运行。
解决：设置 HF_ENDPOINT=https://hf-mirror.com。经 curl 直连测试，hf-mirror.com 返回 200，可达。

问题 2：PowerShell 默认走已宕机的系统代理
现象：用 Invoke-WebRequest 测试时所有站点均"无法连接到远程服务器"。
原因：Invoke-WebRequest 默认继承 IE 代理设置（127.0.0.1:7890），而该代理未运行。
解决：改用 curl --noproxy "*" 直连测试，确认真实可达性；Python 的 requests 不走 IE 代理，故直接设置 HF_ENDPOINT 即可。

问题 3：symlink 警告
现象：huggingface_hub 提示 D:\aiqg\hf_cache 不支持符号链接，缓存为降级模式。
原因：Windows 未开开发者模式，普通用户无符号链接权限。
影响：仅多占磁盘，不影响功能。可接受。

问题 4：后台任务退出码显示为 1
现象：探针后台运行显示 failed exit 1。
原因：tqdm 进度条写入 stderr，被 PowerShell 当作 NativeCommandError。
解决：干净重跑并以 $LASTEXITCODE 判定，实际 exit code = 0，全部断言通过。
