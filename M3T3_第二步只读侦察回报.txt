# ============================================================
# feature_encoder.py · V4.2 · 特征编码
# 规格：5.5.1 / 5.5.2
# 输出：x (8维) + x_fast (4维)
# ============================================================

import numpy as np

from config import DANGER_KEYWORDS, POSITIVE_WORDS, NEGATIVE_WORDS


def safe_cosine(a, b):
    if np.linalg.norm(a) < 1e-8 or np.linalg.norm(b) < 1e-8:
        return 0.0
    return float(np.dot(a, b) / (np.linalg.norm(a) * np.linalg.norm(b)))


def punctuation_heuristic(raw_text):
    count = (raw_text.count("!") + raw_text.count("！") +
             raw_text.count("?") + raw_text.count("？"))
    return min(1.0, count / 3.0)


def pos_count(raw_text):
    return sum(raw_text.count(w) for w in POSITIVE_WORDS)


def neg_count(raw_text):
    return sum(raw_text.count(w) for w in NEGATIVE_WORDS)


def emotion_polarity(x_fast_1):
    return float(x_fast_1)


def person_overlap_ratio(current_event, memory_graph):
    if memory_graph.is_empty():
        return 0.0
    cur_persons = set(current_event.nodes.get("person", []))
    if not cur_persons:
        return 0.0
    mem_persons = set()
    for mem in memory_graph.get_all():
        mem_persons |= set(mem.nodes.get("person", []))
    if not mem_persons:
        return 0.0
    return len(cur_persons & mem_persons) / (len(cur_persons) + 1e-6)


def task_relevance(current_event):
    return 1.0 if current_event.nodes.get("task") else 0.0


def encode(event, M_prev, HabituationSignal_prev, B, memory_graph):
    raw_text = event.raw_text

    x_fast_0 = 1.0 if any(kw in raw_text for kw in DANGER_KEYWORDS) else 0.0
    x_fast_1 = (neg_count(raw_text) - pos_count(raw_text)) / (
        pos_count(raw_text) + neg_count(raw_text) + 1e-6
    )
    x_fast_2 = punctuation_heuristic(raw_text)
    x_fast_3 = 0.0
    x_fast = np.array([x_fast_0, x_fast_1, x_fast_2, x_fast_3])

    x0 = safe_cosine(event.event_embedding, M_prev)
    x1 = person_overlap_ratio(event, memory_graph)
    x2 = task_relevance(event)
    x3 = emotion_polarity(x_fast[1])
    x4 = event.event_severity
    x5 = HabituationSignal_prev
    x6 = B[1]
    x7 = B[2]
    x = np.array([x0, x1, x2, x3, x4, x5, x6, x7], dtype=float)

    return x, x_fast

任务 3：slow_path.py 完整原文

# ============================================================
# slow_path.py · V4.2 · 模块5 慢通路
# 规格：5.5.8
# ============================================================

import numpy as np

from config import w_s, u_s, W_v, U_v, v_s, v_v, b_v, b_s, ETA, W_H


def sigmoid(z):
    return 1.0 / (1.0 + np.exp(-z))


def run_slow_path(x, ReframedContext, M, B, w_PFC, HabituationSignal):
    z_s = (
        w_s @ (x + ReframedContext)
        + u_s @ M
        + v_s @ B
        + ETA * w_PFC
        + W_H * HabituationSignal
        + b_s
    )
    D_slow = float(sigmoid(z_s))

    z_v = W_v @ (x + ReframedContext) + U_v @ M + v_v * D_slow + b_v
    V = sigmoid(z_v)
    return D_slow, V

问 1：x 是怎么算出来的？feature_encoder.encode() 返回的 x，是不是直接等于 event.event_embedding？还是经过变换？

答：x 不等于 event_embedding，x 是 8 个手工标量特征拼成的 8 维特征向量，与 embedding 没有维度绑定关系。

feature_encoder.py 第 65~73 行：

  x1 = person_overlap_ratio(event, memory_graph)       —— 人物重叠比例
  x2 = task_relevance(event)                           —— 任务相关，二值 0/1
  x3 = emotion_polarity(x_fast[1]) = x_fast[1]         —— 情绪极性
  x4 = event.event_severity                            —— 事件严重度
  x5 = HabituationSignal_prev                          —— 习惯化信号
  x6 = B[1]                                            —— 唤醒度分量
  x7 = B[2]                                            —— B 的第三分量

结论：event_embedding 只通过 x0 影响 x，且被 cosine 相似度降维成一个标量。x 的 8 维是"特征维"（相似度、人物、任务、极性、严重度、习惯化、B 分量），不是"嵌入维"。

关键推论：embedding 从 8 变 1024，x 仍然可以是 8 维——只要 x0 的 cosine 两边同维即可。x 的维度与 embedding 的维度是两个独立的事。

问 5：ReframedContext 怎么算的？它是不是也变成 1024 维了？

答：ReframedContext 是 x 的逐元素变换，维度始终等于 x 的维度。

hippocampal_evaluator.py：
  空历史分支（第 24 行）：ReframedContext = np.tanh(x)
  有历史分支（第 52 行）：ReframedContext = np.tanh(x + HistoricalResidual + HabituationSignal)
  其中 HistoricalResidual、HabituationSignal 都是标量。

np.tanh 是逐元素函数，输出形状 = 输入形状。所以：
  x 是 8 维  → ReframedContext 是 8 维（现状）
  x 是 1024 维 → ReframedContext 是 1024 维

结论：ReframedContext 不引入新维度，它跟随 x。若 x 变 1024，它同步变 1024，不是"另外"变成 1024。


问 6：如果 x 从 8 维变 1024 维，哪些地方会立刻出错？逐条列出（文件 + 行 + 原因）。

答：逐条如下。注意区分两类：一类是"x 变 1024"直接导致；一类是"embedding 变 1024 而 M/x0 未同步"导致。

问 8：config.py 里 w_s、u_s、W_v、U_v 是与 x 点积的。如果 x 变 1024 维，这四个参数的维度该怎么变？

答：需要分清这四个参数各自乘的是 x 还是 M，不能一概而论。

slow_path.py 的用法：
  第 17 行  w_s @ (x + ReframedContext)          —— 乘 x
  第 18 行  u_s @ M                              —— 乘 M
  第 26 行  W_v @ (x + ReframedContext)          —— 乘 x
  第 26 行  U_v @ M                              —— 乘 M

因此：
  w_s：(8,) → 必须等于 x 的维度。x 变 1024，w_s 变 (1024,)。
  W_v：(2,8) → 第二维必须等于 x 的维度。x 变 1024，W_v 变 (2,1024)。
  u_s：(8,) → 必须等于 M 的维度。M 变不变取决于 embedding，不取决于 x。
  U_v：(2,8) → 第二维必须等于 M 的维度。同上。

关键点：x 和 M 是两个独立维度。
  现状 x=8、M=8 只是巧合（config.py 第 24 行 n=m=8）。
  若只把 x 变 1024 而 M 仍 8：w_s 需 (1024,)、W_v 需 (2,1024)，而 u_s 仍 (8,)、U_v 仍 (2,8)。
  若 x 与 M 都随嵌入变 1024：四个参数全部变 1024。

附加影响：这四行是 np.random.randn 随机初始化，且 config.py 第 40 行注释声明"与探针 probe_numeric.py 一致"。改维度后随机数序列与数值都会变，会打破 config 与 probe_numeric 的一致性（tests/test_reproducibility.py 第 16、20 行也在比对 w_s）。


问 9：hippocampal_evaluator.py 的 match_score 用的是 cosine 相似度——如果两边都是 1024 维，它会自动工作吗？还是需要改？

答：会自动工作，不需要改。

xinguang.py 第 34~38 行：
  def match_score(current_event_embedding, mem_embedding):
      if np.linalg.norm(current_event_embedding) < 1e-8 or np.linalg.norm(mem_embedding) < 1e-8:
          return 0.0
      return float(np.dot(current_event_embedding, mem_embedding) /
                   (np.linalg.norm(current_event_embedding) * np.linalg.norm(mem_embedding)))

np.dot 与 np.linalg.norm 对任意等长维度都成立。只要两边维度一致（都 1024），自动得到正确的 cosine，无需修改。

调用点：
  hippocampal_evaluator.py 第 17 行  match_score(current_event.event_embedding, mem.event_embedding) > MATCH_THRESHOLD
  xinguang.py 第 55 行  semantic_sim_term = match_score(current_event.event_embedding, mem.event_embedding)

唯一前提：两边必须同维。若一边 8 一边 1024，np.dot 会 ValueError（这正是问 6 第 5、6 条的出错机制）。

同理，hippocampal_evaluator.py 第 49 行 M = np.mean([...], axis=0) 自动跟随嵌入维度，本身也无需改；但 M 传给 slow_path 第 18 行 u_s @ M 时，u_s 的维度必须与 M 一致。
